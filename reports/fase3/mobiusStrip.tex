\documentclass[tikz,border=10pt]{standalone}
\usepackage{tikz}
\usepackage{tikz-3dplot}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta}

\begin{document}
	
	\tdplotsetmaincoords{55}{120}
	\begin{tikzpicture}[tdplot_main_coords, scale=2.5]
		
		\def\R{1.0}
		\def\W{0.25}
		
		% Linhas transversais (ao longo da largura) em posicoes fixas de u
		\foreach \ui in {0,15,...,345} {
			\draw[gray!35, very thin]
			plot[domain=-\W:\W, samples=2, variable=\v]
			({(\R + \v*cos(\ui/2))*cos(\ui)},
			{(\R + \v*cos(\ui/2))*sin(\ui)},
			{\v*sin(\ui/2)});
		}
		
		% Linhas longitudinais em v=-W, v=0, v=+W
		\foreach \vval in {-0.25, 0.0, 0.25} {
			\draw[gray!35, very thin]
			plot[domain=0:360, samples=90, smooth, variable=\u]
			({(\R + \vval*cos(\u/2))*cos(\u)},
			{(\R + \vval*cos(\u/2))*sin(\u)},
			{\vval*sin(\u/2)});
		}
		
		% Bordas destacadas
		\draw[blue!60, thick]
		plot[domain=0:360, samples=120, smooth, variable=\u]
		({(\R - \W*cos(\u/2))*cos(\u)},
		{(\R - \W*cos(\u/2))*sin(\u)},
		{-\W*sin(\u/2)});
		
		\draw[red!60, thick]
		plot[domain=0:360, samples=120, smooth, variable=\u]
		({(\R + \W*cos(\u/2))*cos(\u)},
		{(\R + \W*cos(\u/2))*sin(\u)},
		{\W*sin(\u/2)});
		
		% Eixos coordenados
		\draw[-Stealth, line width=1pt, blue!70!black] (0,0,0) -- (1.6,0,0) node[right] {$z$};
		\draw[-Stealth, line width=1pt, green!70!black] (0,0,0) -- (0,0,0.9) node[above] {$y$};
		\draw[-Stealth, line width=1pt, red!70!black]   (0,0,0) -- (0,1.6,0) node[above] {$x$};
		
		\fill (0,0,0) circle (0.015) node[below left, font=\footnotesize] {$(0,0,0)$};
		
		% Circulo central de referencia
		\draw[dashed, gray!50, thick]
		plot[domain=0:360, samples=60, smooth, variable=\t]
		({\R*cos(\t)}, {\R*sin(\t)}, 0);
		
		% Raio r
		\draw[<->, thick] (0,0,0) -- (\R, 0, 0);
		\node[above, font=\footnotesize] at (0.55, 0.12, 0) {$r$};
		
		% Quadrilatero destacado
		\draw[orange!80, line width=1.5pt]
		(0.392, 0.679, -0.125) -- (0.608, 1.054, 0.125) --
		(0.311, 1.158, 0.152) -- (0.208, 0.775, -0.152) -- cycle;
		
		\fill[black] (0.392, 0.679, -0.125) circle (0.025);
		\node[below, font=\small] at (0.392, 0.679, -0.125) {$(x_1,y_1,z_1)$};
		
		\fill[black] (0.608, 1.054, 0.125) circle (0.025);
		\node[above, font=\small] at (0.608, 1.054, 0.125) {$(x_2,y_2,z_2)$};
		
		\fill[black] (0.208, 0.775, -0.152) circle (0.025);
		\node[below right, font=\small] at (0.208, 0.775, -0.152) {$(x_3,y_3,z_3)$};
		
		\fill[black] (0.311, 1.158, 0.152) circle (0.025);
		\node[above right, font=\small] at (0.311, 1.158, 0.152) {$(x_4,y_4,z_4)$};
		
		% Angulo u
		\draw[purple!70, thick, -{Stealth}]
		plot[domain=0:60, samples=30, variable=\t]
		({0.6*cos(\t)}, {0.6*sin(\t)}, 0);
		\node[purple!70!black, font=\normalsize] at (0.32, 0.38, 0) {$u$};
		
		% Largura 2*width no ponto u=0
		\draw[<->, thick, teal] (0.75, 0, 0) -- (1.25, 0, 0);
		\node[teal, above, font=\footnotesize] at (1.0, 0.06, 0) {$2\,\text{width}$};
		
		% Caixa de anotacoes -- colocada à direita em baixo, fora da fita
		\node[align=left, font=\normalsize, draw, fill=white, rounded corners]
		at (2.2, -1.2, 0) {
			$u \in [0, 2\pi]$ (rota\c{c}\~ao ao longo do anel) \\
			$v \in [-\text{width},\,+\text{width}]$ (largura) \\[4pt]
			$\Delta u = \dfrac{2\pi}{\text{slices}}$ \\[6pt]
			$R = r$ (raio central)
		};
		
	\end{tikzpicture}
	
\end{document}
